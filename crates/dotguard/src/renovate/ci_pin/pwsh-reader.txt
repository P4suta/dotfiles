$ErrorActionPreference = "Stop"

  Where-Object { $_ -cmatch '^rust\s*=\s*"[0-9]+\.[0-9]+(?:\.[0-9]+)?"\s*$' }
if (@($line).Count -ne 1 -or $line -cnotmatch '"([^"]+)"') {
  throw "controller/mise.toml must contain exactly one reviewable Rust pin."
}
